package com.p2psecure.share

import android.app.Activity
import android.content.res.ColorStateList
import android.graphics.Color
import android.view.View
import android.view.ViewGroup
import android.widget.Button
import android.widget.EditText
import android.widget.ProgressBar
import android.widget.TextView
import androidx.core.view.WindowCompat

object ThemeHelper {
    data class Palette(val bg: Int, val card: Int, val accent: Int, val text: Int, val muted: Int)

    fun palette(name: String): Palette = when (name) {
        "blue" -> Palette(
            bg = Color.parseColor("#071428"),
            card = Color.parseColor("#0E2A52"),
            accent = Color.parseColor("#3D8BFF"),
            text = Color.parseColor("#EAF3FF"),
            muted = Color.parseColor("#8FB4E8")
        )
        "green" -> Palette(
            bg = Color.parseColor("#10221A"),
            card = Color.parseColor("#163127"),
            accent = Color.parseColor("#198754"),
            text = Color.parseColor("#E9F7EF"),
            muted = Color.parseColor("#8FB9A4")
        )
        else -> Palette(
            bg = Color.parseColor("#1A1A2E"),
            card = Color.parseColor("#16213E"),
            accent = Color.parseColor("#0D6EFD"),
            text = Color.parseColor("#E8EEFC"),
            muted = Color.parseColor("#8EA0C8")
        )
    }

    fun apply(activity: Activity, root: View, themeName: String) {
        val p = palette(themeName)
        try {
            activity.window.statusBarColor = p.bg
            activity.window.navigationBarColor = p.bg
            activity.window.setBackgroundDrawable(android.graphics.drawable.ColorDrawable(p.bg))
            WindowCompat.getInsetsController(activity.window, root)
                .isAppearanceLightStatusBars = false
        } catch (_: Exception) {
        }
        if (root.id != R.id.monitorPanel) root.setBackgroundColor(p.bg)
        try {
            tint(root, p)
        } catch (_: Exception) {
        }
    }

    private fun tint(view: View, p: Palette) {
        when (view) {
            is Button -> {
                view.backgroundTintList = ColorStateList.valueOf(p.accent)
                view.setTextColor(Color.WHITE)
            }
            is EditText -> {
                view.setBackgroundColor(p.card)
                view.setTextColor(p.text)
                view.setHintTextColor(p.muted)
            }
            is ProgressBar -> {
                view.progressTintList = ColorStateList.valueOf(p.accent)
                view.progressBackgroundTintList = ColorStateList.valueOf(p.card)
            }
            is TextView -> {
                val muted = view.tag == "muted"
                view.setTextColor(if (muted) p.muted else p.text)
            }
        }
        if (view.id == R.id.monitorPanel) view.setBackgroundColor(p.card)
        if (view is ViewGroup) {
            for (i in 0 until view.childCount) tint(view.getChildAt(i), p)
        }
    }
}

package com.p2psecure.share

import android.content.Context
import android.net.ConnectivityManager
import android.net.NetworkCapabilities
import android.net.wifi.WifiManager
import android.text.format.Formatter
import java.net.Inet4Address
import java.net.NetworkInterface

object NetworkHelper {
    fun onWifi(context: Context): Boolean {
        return try {
            val cm = context.getSystemService(Context.CONNECTIVITY_SERVICE) as ConnectivityManager
            val net = cm.activeNetwork ?: return false
            val caps = cm.getNetworkCapabilities(net) ?: return false
            caps.hasTransport(NetworkCapabilities.TRANSPORT_WIFI)
        } catch (_: Exception) {
            false
        }
    }

    fun onCellular(context: Context): Boolean {
        val cm = context.getSystemService(Context.CONNECTIVITY_SERVICE) as ConnectivityManager
        val net = cm.activeNetwork ?: return false
        val caps = cm.getNetworkCapabilities(net) ?: return false
        return caps.hasTransport(NetworkCapabilities.TRANSPORT_CELLULAR)
    }

    fun wifiIpv4(context: Context): String {
        try {
            val wm = context.applicationContext.getSystemService(Context.WIFI_SERVICE) as WifiManager
            val ip = Formatter.formatIpAddress(wm.connectionInfo.ipAddress)
            if (ip.isNotBlank() && ip != "0.0.0.0") return ip
        } catch (_: Exception) {
        }
        return localIpv4()
    }

    fun localIpv4(): String {
        val interfaces = NetworkInterface.getNetworkInterfaces()?.toList() ?: return ""
        fun pick(pred: (NetworkInterface) -> Boolean): String {
            for (nif in interfaces) {
                if (!nif.isUp || nif.isLoopback || !pred(nif)) continue
                for (addr in nif.inetAddresses) {
                    if (!addr.isLoopbackAddress && addr is Inet4Address) {
                        val ip = addr.hostAddress ?: continue
                        if (!ip.startsWith("169.254")) return ip
                    }
                }
            }
            return ""
        }
        val wifi = pick {
            it.name.startsWith("wlan") || it.name.startsWith("ap") || it.name.startsWith("swlan")
        }
        if (wifi.isNotBlank()) return wifi
        return pick { true }
    }
}

package com.p2psecure.share

import android.util.Log
import java.text.SimpleDateFormat
import java.util.Date
import java.util.Locale
import java.util.concurrent.CopyOnWriteArrayList

object AppLog {
    private const val MAX = 400
    private val lines = CopyOnWriteArrayList<String>()
    private val fmt = SimpleDateFormat("HH:mm:ss", Locale.US)

    fun i(msg: String) = add("I", msg)
    fun w(msg: String) = add("W", msg)
    fun e(msg: String, t: Throwable? = null) {
        add("E", if (t != null) "$msg ${t.javaClass.simpleName}: ${t.message}" else msg)
        if (t != null) Log.e(JavaTunnel.TAG, msg, t) else Log.e(JavaTunnel.TAG, msg)
    }

    @Synchronized
    private fun add(level: String, msg: String) {
        val line = "${fmt.format(Date())} $level $msg"
        lines.add(line)
        while (lines.size > MAX) lines.removeAt(0)
        when (level) {
            "W" -> Log.w(JavaTunnel.TAG, msg)
            "E" -> {}
            else -> Log.i(JavaTunnel.TAG, msg)
        }
    }

    fun snapshot(): String = lines.takeLast(180).joinToString("\n")

    fun clear() {
        lines.clear()
    }
}

package com.p2psecure.share

import android.util.Base64
import org.json.JSONObject
import java.math.BigInteger
import java.security.KeyFactory
import java.security.KeyPairGenerator
import java.security.SecureRandom
import java.security.interfaces.ECPublicKey
import java.security.spec.ECGenParameterSpec
import java.security.spec.ECParameterSpec
import java.security.spec.ECPoint
import java.security.spec.ECPublicKeySpec
import javax.crypto.Cipher
import javax.crypto.KeyAgreement
import javax.crypto.spec.GCMParameterSpec
import javax.crypto.spec.SecretKeySpec

class CryptoEngine {
    private val pair = KeyPairGenerator.getInstance("EC").run {
        initialize(ECGenParameterSpec("secp256r1"))
        generateKeyPair()
    }
    private var aes: SecretKeySpec? = null

    fun pubJwk(): String {
        val pub = pair.public as ECPublicKey
        val w = pub.w
        return JSONObject()
            .put("kty", "EC")
            .put("crv", "P-256")
            .put("x", b64url(to32(w.affineX)))
            .put("y", b64url(to32(w.affineY)))
            .toString()
    }

    fun derive(peerJwkRaw: String) {
        val jwk = JSONObject(peerJwkRaw)
        val x = BigInteger(1, b64urlDecode(jwk.getString("x")))
        val y = BigInteger(1, b64urlDecode(jwk.getString("y")))
        val params = (pair.public as ECPublicKey).params
        val spec = ECPublicKeySpec(ECPoint(x, y), params)
        val peer = KeyFactory.getInstance("EC").generatePublic(spec)
        val ka = KeyAgreement.getInstance("ECDH")
        ka.init(pair.private)
        ka.doPhase(peer, true)
        val shared = ka.generateSecret()
        val key = hkdf(shared, ByteArray(32), "p2p-v1".toByteArray(), 32)
        aes = SecretKeySpec(key, "AES")
    }

    fun encrypt(data: ByteArray, nonce: ByteArray): ByteArray {
        val c = Cipher.getInstance("AES/GCM/NoPadding")
        c.init(Cipher.ENCRYPT_MODE, aes, GCMParameterSpec(128, nonce))
        return c.doFinal(data)
    }

    fun ready(): Boolean = aes != null

    companion object {
        fun randomNonce(): ByteArray {
            val n = ByteArray(12)
            SecureRandom().nextBytes(n)
            return n
        }

        private fun to32(n: BigInteger): ByteArray {
            var b = n.toByteArray()
            if (b.size > 32) b = b.copyOfRange(b.size - 32, b.size)
            if (b.size < 32) b = ByteArray(32 - b.size) + b
            return b
        }

        private fun b64url(data: ByteArray): String =
            Base64.encodeToString(data, Base64.URL_SAFE or Base64.NO_WRAP or Base64.NO_PADDING)

        private fun b64urlDecode(text: String): ByteArray {
            var t = text.replace('-', '+').replace('_', '/')
            while (t.length % 4 != 0) t += "="
            return Base64.decode(t, Base64.DEFAULT)
        }

        private fun hkdf(ikm: ByteArray, salt: ByteArray, info: ByteArray, length: Int): ByteArray {
            val mac = javax.crypto.Mac.getInstance("HmacSHA256")
            mac.init(SecretKeySpec(salt, "HmacSHA256"))
            val prk = mac.doFinal(ikm)
            mac.init(SecretKeySpec(prk, "HmacSHA256"))
            mac.update(info)
            mac.update(0x01)
            return mac.doFinal().copyOf(length)
        }
    }
}

package com.p2psecure.share

import com.jcraft.jsch.Channel
import com.jcraft.jsch.ChannelShell
import com.jcraft.jsch.JSch
import com.jcraft.jsch.Session
import java.io.ByteArrayOutputStream
import java.net.URI
import java.util.Properties

data class TunnelTarget(
    val name: String,
    val user: String,
    val host: String,
    val port: Int,
    val remotePort: Int
)

class JavaTunnel(val target: TunnelTarget) {
    private var session: Session? = null
    private var shell: Channel? = null
    @Volatile private var watchRunning = false
    private var watchThread: Thread? = null
    @Volatile var publicUrl: String = ""
        private set
    @Volatile var lastError: String = ""
        private set
    @Volatile var lastBanner: String = ""
        private set
    @Volatile var startedAt: Long = 0
        private set
    @Volatile var rttMs: Long = -1
        private set

    fun isAlive(): Boolean {
        val s = session
        return s != null && s.isConnected && publicUrl.isNotBlank() && isTunnelUrl(publicUrl)
    }

    @Synchronized
    fun start(localPort: Int): String {
        if (isAlive()) {
            AppLog.i("reuse ${target.name} $publicUrl")
            return publicUrl
        }
        stop()
        lastError = ""
        publicUrl = ""
        AppLog.i("try ${target.name} ${target.user}@${target.host}:${target.port}")
        val url = tryOne(localPort)
        if (url.isNotBlank() && isTunnelUrl(url)) {
            publicUrl = url
            lastError = ""
            startedAt = System.currentTimeMillis()
            AppLog.i("tunnel ready ${target.name} $url")
            startWatchdog()
            return url
        }
        AppLog.w("skip ${target.name} err=$lastError")
        disconnectQuiet()
        if (lastError.isBlank()) lastError = "no_url_${target.name}"
        return ""
    }

    @Synchronized
    fun stop() {
        watchRunning = false
        try {
            watchThread?.interrupt()
        } catch (_: Exception) {
        }
        watchThread = null
        disconnectQuiet()
        publicUrl = ""
    }

    private fun disconnectQuiet() {
        try {
            shell?.disconnect()
        } catch (_: Exception) {
        }
        try {
            session?.disconnect()
        } catch (_: Exception) {
        }
        shell = null
        session = null
    }

    private fun tryOne(localPort: Int): String {
        var sess: Session? = null
        var ch: Channel? = null
        return try {
            val jsch = JSch()
            sess = jsch.getSession(target.user, target.host, target.port)
            sess.setPassword("")
            val cfg = Properties()
            cfg["StrictHostKeyChecking"] = "no"
            cfg["PreferredAuthentications"] = "none,password,keyboard-interactive,publickey"
            cfg["UseDNS"] = "no"
            sess.setConfig(cfg)
            sess.serverAliveInterval = 12000
            sess.serverAliveCountMax = 16
            sess.setDaemonThread(true)
            sess.connect(16000)
            try {
                sess.setPortForwardingR("*", target.remotePort, "127.0.0.1", localPort)
            } catch (e: Exception) {
                AppLog.w("rforward * ${target.name} ${e.message}")
                sess.setPortForwardingR(target.remotePort, "127.0.0.1", localPort)
            }
            val pair = captureUrl(sess)
            val url = pair.first
            ch = pair.second
            lastBanner = pair.third
            if (url.isBlank() || !isTunnelUrl(url)) {
                lastError = "no_url_${target.name}"
                AppLog.w("no tunnel url from ${target.name}")
                ch?.disconnect()
                sess.disconnect()
                return ""
            }
            session = sess
            shell = ch
            url.trimEnd('/')
        } catch (e: Exception) {
            lastError = (e.message ?: target.name).take(120)
            AppLog.e("tryOne ${target.name} failed", e)
            try {
                ch?.disconnect()
            } catch (_: Exception) {
            }
            try {
                sess?.disconnect()
            } catch (_: Exception) {
            }
            ""
        }
    }

    private fun captureUrl(sess: Session): Triple<String, Channel?, String> {
        val channel = sess.openChannel("shell") as ChannelShell
        channel.setPty(false)
        val input = channel.inputStream
        channel.connect(8000)
        val baos = ByteArrayOutputStream()
        val buf = ByteArray(4096)
        val end = System.currentTimeMillis() + 22000
        var url = ""
        var text = ""
        while (System.currentTimeMillis() < end && sess.isConnected) {
            val avail = try {
                input.available()
            } catch (_: Exception) {
                0
            }
            if (avail > 0) {
                val n = input.read(buf, 0, buf.size.coerceAtMost(avail))
                if (n > 0) baos.write(buf, 0, n)
            }
            text = baos.toString("UTF-8")
            url = extractUrl(text)
            if (url.isNotBlank()) break
            Thread.sleep(250)
        }
        if (url.isBlank()) {
            text = baos.toString("UTF-8")
            url = extractUrl(text)
        }
        AppLog.i("banner ${target.name}=${text.replace('\n', ' ').take(240)}")
        AppLog.i("extracted ${target.name}=$url")
        return Triple(url, channel, text)
    }

    private fun startWatchdog() {
        watchRunning = true
        watchThread = Thread {
            val buf = ByteArray(4096)
            while (watchRunning) {
                try {
                    val s = session
                    val ch = shell
                    if (s == null || !s.isConnected) {
                        AppLog.w("tunnel dropped ${target.name}")
                        publicUrl = ""
                        lastError = "tunnel_dropped"
                        break
                    }
                    if (ch != null && ch.isConnected) {
                        val ins = try {
                            ch.inputStream
                        } catch (_: Exception) {
                            null
                        }
                        val avail = try {
                            ins?.available() ?: 0
                        } catch (_: Exception) {
                            0
                        }
                        if (ins != null && avail > 0) {
                            val n = ins.read(buf, 0, buf.size.coerceAtMost(avail))
                            if (n > 0) {
                                val t = String(buf, 0, n, Charsets.UTF_8)
                                AppLog.i("ssh ${target.name} ${t.replace('\n', ' ').take(120)}")
                                val low = t.lowercase()
                                if (low.contains("closed") || low.contains("expired") ||
                                    low.contains("offline") || low.contains("denied")
                                ) {
                                    publicUrl = ""
                                    lastError = "tunnel_closed"
                                    try {
                                        s.disconnect()
                                    } catch (_: Exception) {
                                    }
                                    break
                                }
                            }
                        }
                    }
                    try {
                        val t0 = System.nanoTime()
                        s.sendKeepAliveMsg()
                        rttMs = ((System.nanoTime() - t0) / 1_000_000L).coerceAtLeast(1)
                    } catch (_: Exception) {
                    }
                    Thread.sleep(4000)
                } catch (_: InterruptedException) {
                    break
                } catch (e: Exception) {
                    AppLog.w("watch ${target.name} ${e.message}")
                    publicUrl = ""
                    lastError = "tunnel_dropped"
                    break
                }
            }
        }.also {
            it.name = "p2p-watch-${target.name}"
            it.isDaemon = true
            it.start()
        }
    }

    private fun extractUrl(text: String): String {
        val found = Regex("https?://[a-zA-Z0-9._\\-]+").findAll(text).map { it.value.trimEnd('/') }
        for (raw in found) {
            val url = if (raw.startsWith("http://")) "https://" + raw.removePrefix("http://") else raw
            if (isTunnelUrl(url)) return url
        }
        return ""
    }

package com.p2psecure.share

import java.util.concurrent.ConcurrentHashMap
import java.util.concurrent.CopyOnWriteArrayList

class TunnelPool {
    private val tunnels = CopyOnWriteArrayList<JavaTunnel>()
    private val connecting = ConcurrentHashMap.newKeySet<String>()
    @Volatile var primaryName: String = ""
        private set

    fun alive(): List<JavaTunnel> = tunnels.filter { it.isAlive() }

    fun aliveUrls(): List<String> = alive().map { it.publicUrl }.distinct()

    fun failover(): JavaTunnel? {
        val live = alive()
        if (live.isEmpty()) {
            primaryName = ""
            return null
        }
        val keep = live.firstOrNull { it.target.name == primaryName }
        if (keep != null) return keep
        val pick = live.minByOrNull { it.rttMs.takeIf { it > 0 } ?: Long.MAX_VALUE } ?: live.first()
        AppLog.i("failover ${primaryName.ifBlank { "-" }} -> ${pick.target.name}")
        primaryName = pick.target.name
        return pick
    }

    fun primary(): JavaTunnel? {
        val keep = alive().firstOrNull { it.target.name == primaryName }
        if (keep != null) return keep
        return failover()
    }

    fun primaryUrl(): String = primary()?.publicUrl.orEmpty()

    fun isAlive(): Boolean = primary()?.isAlive() == true

    fun lastError(): String {
        return tunnels.map { it.lastError }.firstOrNull { it.isNotBlank() }.orEmpty()
    }

    fun connectingNames(): Set<String> = connecting.toSet()

    fun snapshot(): List<TunnelRow> {
        val now = System.currentTimeMillis()
        val live = tunnels.associateBy { it.target.name }
        return JavaTunnel.PROVIDERS.map { p ->
            val t = live[p.name]
            when {
                t != null && t.isAlive() -> TunnelRow(
                    name = p.name,
                    state = "up",
                    url = t.publicUrl,
                    primary = p.name == primaryName,
                    ageSec = if (t.startedAt > 0) (now - t.startedAt) / 1000 else 0,
                    rttMs = t.rttMs
                )
                p.name in connecting -> TunnelRow(p.name, "opening", "", false, 0, -1)
                else -> TunnelRow(p.name, "down", "", false, 0, -1)
            }
        }
    }

    fun ensure(localPort: Int): String {
        pruneDead()
        startMissing(localPort)
        val deadline = System.currentTimeMillis() + 28000
        while (System.currentTimeMillis() < deadline) {
            val url = primaryUrl()
            if (url.isNotBlank()) {
                AppLog.i("pool primary=${primaryName} urls=${aliveUrls()}")
                return url
            }
            Thread.sleep(300)
        }
        val url = primaryUrl()
        AppLog.i("pool wait done primary=${primaryName} urls=${aliveUrls()}")
        return url
    }

    fun refresh(localPort: Int): String {
        pruneDead()
        startMissing(localPort)
        return primaryUrl()
    }

    fun stop() {
        for (t in tunnels) {
            try {
                t.stop()
            } catch (_: Exception) {
            }
        }
        tunnels.clear()
        connecting.clear()
        primaryName = ""
    }

    private fun pruneDead() {
        val dead = tunnels.filter { !it.isAlive() }
        for (t in dead) {
            AppLog.w("pool drop ${t.target.name}")
            try {
                t.stop()
            } catch (_: Exception) {
            }
            tunnels.remove(t)
            connecting.remove(t.target.name)
            if (primaryName == t.target.name) primaryName = ""
        }
    }

    private fun startMissing(localPort: Int) {
        val liveNames = tunnels.filter { it.isAlive() }.map { it.target.name }.toSet()
        for (p in JavaTunnel.PROVIDERS) {
            if (p.name in liveNames) continue
            if (!connecting.add(p.name)) continue
            Thread {
                try {
                    val jt = JavaTunnel(p)
                    val url = jt.start(localPort)
                    if (url.isNotBlank()) {
                        tunnels.add(jt)
                        AppLog.i("pool add ${p.name} $url n=${alive().size}")
                    } else {
                        try {
                            jt.stop()
                        } catch (_: Exception) {
                        }
                    }
                } catch (e: Exception) {
                    AppLog.e("pool ${p.name}", e)
                } finally {
                    connecting.remove(p.name)
                }
            }.apply {
                name = "p2p-open-${p.name}"
                isDaemon = true
                start()
            }
        }
    }
}

    class Front(private val context: Context, port: Int) : NanoWSD("0.0.0.0", port) {
        private val clients = ConcurrentHashMap<String, Socket>()
        private val pending = ConcurrentHashMap<String, MutableList<String>>()

        override fun openWebSocket(handshake: IHTTPSession): WebSocket = Socket(handshake)

        override fun serveHttp(session: IHTTPSession): Response {
            if (session.method == Method.OPTIONS) {
                val res = newFixedLengthResponse(Response.Status.OK, NanoHTTPD.MIME_PLAINTEXT, "")
                cors(res)
                return res
            }
            val uri = session.uri ?: "/"
            if (uri == "/favicon.ico") {
                val res = newFixedLengthResponse(Response.Status.NO_CONTENT, NanoHTTPD.MIME_PLAINTEXT, "")
                cors(res)
                return res
            }
            if (uri == "/tunnels.json" || uri == "/api/tunnels") {
                val o = JSONObject()
                val arr = JSONArray()
                ShareRuntime.wsUrls().forEach { arr.put(it) }
                o.put("ws", arr)
                o.put("link", ShareRuntime.receiverLink)
                val res = newFixedLengthResponse(Response.Status.OK, "application/json", o.toString())
                cors(res)
                res.addHeader("Cache-Control", "no-store")
                return res
            }
            var html = context.assets.open("receiver.html").bufferedReader().use { it.readText() }
            val arr = JSONArray()
            ShareRuntime.wsUrls().forEach { arr.put(it) }
            html = html.replace("const BACKUP_WS = [];", "const BACKUP_WS = $arr;")
            val res = newFixedLengthResponse(Response.Status.OK, "text/html; charset=utf-8", html)
            cors(res)
            res.addHeader("Cache-Control", "no-store")
            res.addHeader("ngrok-skip-browser-warning", "1")
            return res
        }

        private fun cors(res: Response) {
            res.addHeader("Access-Control-Allow-Origin", "*")
            res.addHeader("Access-Control-Allow-Headers", "*")
            res.addHeader("Access-Control-Allow-Methods", "GET,POST,OPTIONS")
        }

        inner class Socket(handshake: IHTTPSession) : WebSocket(handshake) {
            private var deviceId: String? = null

            override fun onOpen() {}

            override fun onClose(
                code: WebSocketFrame.CloseCode?,
                reason: String?,
                initiatedByRemote: Boolean
            ) {
                val id = deviceId
                if (id != null) clients.remove(id, this)
                ConnStats.clients = clients.size
            }

            override fun onPong(pong: WebSocketFrame?) {}

            override fun onException(exception: IOException) {}

            override fun onMessage(message: WebSocketFrame) {
                val msg = JSONObject(message.textPayload)
                when (msg.optString("type")) {
                    "register" -> {
                        val id = normalizeId(
                            msg.optString("device_id").ifBlank {
                                java.lang.Long.toHexString(SecureRandom().nextLong())
                            }
                        )
                        deviceId = id
                        clients[id] = this
                        ConnStats.clients = clients.size
                        AppLog.i("ws register $id clients=${clients.size}")
                        send(JSONObject().put("type", "registered").put("device_id", id).toString())
                        pending.remove(id)?.forEach { send(it) }
                    }
                    "ping" -> send(JSONObject().put("type", "pong").toString())
                    "chunk_nack", "resume" -> {
                        val sender = deviceId ?: return
                        val target = normalizeId(msg.optString("target_id"))
                        if (target.isBlank()) return
                        val payload = JSONObject(msg.toString())
                        payload.put("from_id", sender)
                        payload.remove("target_id")
                        val dest = clients[target]
                        if (dest != null && dest.isOpen) dest.send(payload.toString())
                        else pending.getOrPut(target) { mutableListOf() }.add(payload.toString())
                    }
                    else -> {
                        val sender = deviceId ?: return
                        val target = normalizeId(msg.optString("target_id"))
                        if (target.isBlank()) return
                        val payload = JSONObject(msg.toString())
                        payload.put("from_id", sender)
                        payload.remove("target_id")
                        val text = payload.toString()
                        val dest = clients[target]
                        val kind = msg.optString("type")
                        if (dest != null && dest.isOpen) {
                            dest.send(text)
                            if (kind != "chunk" && kind != "chunk_ack") {
                                AppLog.i("ws fwd $kind $sender -> $target")
                            }
                        } else if (kind != "chunk" && kind != "chunk_ack") {
                            pending.getOrPut(target) { mutableListOf() }.add(text)
                            AppLog.w("ws queue $kind $sender -> $target")
                        }
                    }
                }
            }
        }
    }
}

object ShareRuntime {
    @Volatile var receiverLink: String = ""
    @Volatile var lastError: String = ""

        }.distinct()
    }

    fun hasInternetLink(): Boolean {
        val link = receiverLink
        val alive = pool?.isAlive() == true
        return alive &&
            link.startsWith("https://") &&
            !link.contains("dashboard.pinggy.io") &&
            !link.contains("console.serveo.net") &&
            JavaTunnel.companionIsTunnel(link.removeSuffix("/receiver.html"))
    }

    @Synchronized
    fun ensureStarted(context: Context) {
        val app = context.applicationContext
        try {
            if (server == null) {
                server = LocalServer(app)
                server?.start()
                AppLog.i("local server :${LocalServer.HTTP_PORT}")
            }
        } catch (e: Exception) {
            lastError = e.message ?: "server"
            AppLog.e("server start failed", e)
            return
        }
        val p = pool ?: TunnelPool().also { pool = it }
        val prev = receiverLink
        val publicBase = if (p.isAlive()) {
            p.refresh(LocalServer.HTTP_PORT)
        } else {
            lastError = ""
            p.ensure(LocalServer.HTTP_PORT)
        }
        if (publicBase.isBlank()) lastError = p.lastError().ifBlank { "tunnel_failed" }
        val next = if (publicBase.isNotBlank() && JavaTunnel.companionIsTunnel(publicBase)) {
            "$publicBase/receiver.html"
        } else {
            ""
        }
        if (prev.isNotBlank() && next.isNotBlank() && prev != next) {
            AppLog.w("primary switched $prev -> $next")
        }
        receiverLink = next
        AppLog.i("receiverLink=$receiverLink ready=${hasInternetLink()} backups=${wsUrls().size} err=$lastError")
    }

    @Synchronized
    fun stop() {
        try {
            pool?.stop()
        } catch (_: Exception) {
        }

        }

package com.p2psecure.share

import android.os.ParcelFileDescriptor
import android.system.Os
import java.security.MessageDigest

class FileSource(
    private val pfd: ParcelFileDescriptor,
    val size: Long,
    val name: String
) {
    fun read(index: Int, chunkSize: Int): ByteArray {
        val offset = index.toLong() * chunkSize
        if (offset >= size) return ByteArray(0)
        val len = minOf(chunkSize.toLong(), size - offset).toInt()
        val buf = ByteArray(len)
        var got = 0
        while (got < len) {
            val n = Os.pread(pfd.fileDescriptor, buf, got, len - got, offset + got)
            if (n <= 0) break
            got += n
        }
        return if (got == len) buf else buf.copyOf(got)
    }

    fun sha256(chunkSize: Int): String {
        val md = MessageDigest.getInstance("SHA-256")
        val buf = ByteArray(chunkSize)
        var off = 0L
        while (off < size) {
            val len = minOf(buf.size.toLong(), size - off).toInt()
            var got = 0
            while (got < len) {
                val n = Os.pread(pfd.fileDescriptor, buf, got, len - got, off + got)
                if (n <= 0) break
                got += n
            }
            if (got <= 0) break
            md.update(buf, 0, got)
            off += got
        }
        return md.digest().joinToString("") { "%02x".format(it) }
    }

    fun close() {
        try {
            pfd.close()
        } catch (_: Exception) {
        }
    }
}

package com.p2psecure.share

import android.util.Base64
import org.java_websocket.client.WebSocketClient
import org.java_websocket.handshake.ServerHandshake
import org.json.JSONArray
import org.json.JSONObject
import java.net.URI
import java.security.MessageDigest
import java.util.concurrent.ConcurrentHashMap
import java.util.concurrent.LinkedBlockingQueue
import java.util.concurrent.TimeUnit

class SenderClient(
    private val relay: String,
    targetRaw: String,
    private val source: FileSource,
    private val onStatus: (String) -> Unit,
    private val onProgress: (Int) -> Unit
) {
    private val target = normalizeId(targetRaw)
    private val senderId = java.lang.Long.toHexString(System.nanoTime()).padStart(16, 'a').takeLast(16)
    private val crypto = CryptoEngine()
    private val incoming = LinkedBlockingQueue<JSONObject>()
    private val acks = ConcurrentHashMap<Int, Boolean>()
    private val ackAt = ConcurrentHashMap<Int, Long>()
    @Volatile private var client: WebSocketClient? = null
    @Volatile private var accepted = false
    @Volatile private var gotKey = false
    private val fileName = source.name
    private val fileSize = source.size
    private val total = if (fileSize <= 0L) 0 else ((fileSize + CHUNK - 1) / CHUNK).toInt()
    private var fileHash = ""

    fun run() {
        ConnStats.reset()
        ConnStats.phase = "connecting"
        ConnStats.fileName = fileName
        ConnStats.fileSize = fileSize
        if (target.length < 8) throw IllegalStateException("bad_id")
        if (fileSize <= 0L || total <= 0) throw IllegalStateException("empty_file")
        val tStart = System.currentTimeMillis()
        var lastMark = tStart
        var lastBytes = 0L
        var lastProgress = tStart
        try {
            AppLog.i("hashing $fileName $fileSize")
            ConnStats.phase = "hashing"
            fileHash = source.sha256(CHUNK)
            AppLog.i("hash $fileHash")
            while (true) {
                try {
                    ensureSocket()
                    waitAccept()
                    ConnStats.phase = "sending"
                    onStatus("sending")
                    pump()
                    val elapsed = (System.currentTimeMillis() - tStart).coerceAtLeast(1)
                    ConnStats.speedBps = fileSize * 1000.0 / elapsed
                    AppLog.i("sent $fileSize bytes in ${elapsed}ms ${ConnStats.speedText()}")
                    sendJson(
                        JSONObject()
                            .put("type", "transfer_complete")
                            .put("target_id", target)
                            .put("final_hash", fileHash)
                    )
                    ConnStats.phase = "done"
                    ConnStats.sentBytes = fileSize
                    onStatus("done")
                    try {
                        client?.close()
                    } catch (_: Exception) {
                    }
                    return
                } catch (e: RejectedException) {
                    ConnStats.phase = "rejected"
                    onStatus("rejected")
                    return
                } catch (e: Exception) {
                    AppLog.w("sender drop ${e.message} — resume")
                    ConnStats.phase = "resume"
                    dropSocket()
                    if (System.currentTimeMillis() - lastProgress > 180000 && acks.isEmpty()) {
                        throw IllegalStateException("stalled")
                    }
                    Thread.sleep(900)
                } finally {
                    val now = System.currentTimeMillis()
                    val dt = now - lastMark
                    if (dt >= 400) {
                        val db = ConnStats.sentBytes - lastBytes
                        if (dt > 0) ConnStats.speedBps = db * 1000.0 / dt
                        lastMark = now
                        lastBytes = ConnStats.sentBytes
                        if (db > 0) lastProgress = now
                    }
                }
            }
        } finally {
            source.close()
        }
    }

    private fun ensureSocket() {
        val cur = client
        if (cur != null && cur.isOpen) return
        dropSocket()
        val headers = mapOf("ngrok-skip-browser-warning" to "1")
        val ws = object : WebSocketClient(URI(relay), headers) {
            override fun onOpen(handshakedata: ServerHandshake) {
                connectionLostTimeout = 0
                send(
                    JSONObject()
                        .put("type", "register")
                        .put("mode", "sender")
                        .put("device_id", senderId)
                        .put("public_key", crypto.pubJwk())
                        .toString()
                )
            }
            override fun onMessage(message: String) {
                val msg = JSONObject(message)
                when (msg.optString("type")) {
                    "chunk_ack" -> {
                        val idx = msg.optInt("index")
                        acks[idx] = true
                        val sent = ackAt.remove(idx)
                        if (sent != null) {
                            ConnStats.rttMs = System.currentTimeMillis() - sent
                            ConnStats.acksOk += 1
                        }
                    }
                    "chunk_nack" -> {
                        val idx = msg.optInt("index")
                        acks.remove(idx)
                        ConnStats.acksRetry += 1
                        AppLog.w("nack $idx")
                        incoming.offer(msg)
                    }
                    "resume" -> incoming.offer(msg)
                    else -> incoming.offer(msg)
                }
            }
            override fun onClose(code: Int, reason: String, remote: Boolean) {
                AppLog.w("sender ws closed $code $reason")
            }
            override fun onError(ex: Exception) {
                AppLog.w("sender ws ${ex.message}")
            }
        }
        AppLog.i("sender connect $relay target=$target")
        if (!ws.connectBlocking(20, TimeUnit.SECONDS) || !ws.isOpen) {
            throw IllegalStateException("ws_connect")
        }
        client = ws
        waitFor("registered", 20)
        sendJson(JSONObject().put("type", "public_key").put("target_id", target).put("public_key", crypto.pubJwk()))
    }

    private fun waitAccept() {
        val pub = crypto.pubJwk()
        fun sendRequest() {
            sendJson(
                JSONObject()
                    .put("type", "transfer_request")
                    .put("target_id", target)
                    .put("filename", fileName)
                    .put("file_size", fileSize)
                    .put("total_chunks", total)
                    .put("chunk_size", CHUNK)
                    .put("file_hash", fileHash)
                    .put("public_key", pub)
            )
            AppLog.i("sender request $fileName $fileSize -> $target acked=${acks.size}")
        }
        sendRequest()
        if (accepted) {
            val drainEnd = System.currentTimeMillis() + 4000
            while (System.currentTimeMillis() < drainEnd) {
                val msg = incoming.poll(200, TimeUnit.MILLISECONDS) ?: continue
                handleControl(msg)
                if (msg.optString("type") == "resume") return
            }
            return
        }
        ConnStats.phase = "waiting"
        onStatus("waiting")
        val waitEnd = System.currentTimeMillis() + 180000
        var lastAsk = System.currentTimeMillis()
        while (!accepted) {
            if (System.currentTimeMillis() > waitEnd) throw IllegalStateException("timeout")
            if (client?.isOpen != true) throw IllegalStateException("ws_closed")
            val msg = incoming.poll(3, TimeUnit.SECONDS)
            if (msg == null) {
                if (System.currentTimeMillis() - lastAsk > 8000) {
                    sendRequest()
                    lastAsk = System.currentTimeMillis()
                }
                continue
            }
            when (msg.optString("type")) {
                "public_key" -> {
                    if (!gotKey) {
                        crypto.derive(msg.getString("public_key"))
                        gotKey = true
                        AppLog.i("sender got peer key")
                    }
                }
                "transfer_response" -> {
                    if (!msg.optBoolean("accepted")) throw RejectedException()
                    if (msg.has("public_key") && !gotKey) {
                        crypto.derive(msg.getString("public_key"))
                        gotKey = true
                    }
                    if (gotKey) accepted = true
                }
                "resume" -> applyResume(msg)
            }
        }
        if (!gotKey) throw IllegalStateException("no_key")
    }

    private fun pump() {
        val window = 32
        while (true) {
            if (client?.isOpen != true) throw IllegalStateException("ws_closed")
            drainControl()
            val missing = ArrayList<Int>()
            var i = 0
            while (i < total) {
                if (acks[i] != true) missing.add(i)
                i++
            }
            if (missing.isEmpty()) {
                onProgress(100)
                return
            }
            val done = total - missing.size
            ConnStats.sentBytes = (done.toLong() * CHUNK).coerceAtMost(fileSize)
            onProgress((done * 100) / total)
            val inFlight = missing.count { ackAt.containsKey(it) && acks[it] != true }
            var sentNow = 0
            for (idx in missing) {
                if (sentNow + maxOf(inFlight, 0) >= window) break
                val last = ackAt[idx]
                if (last != null && System.currentTimeMillis() - last < 2500) continue
                if (last != null) ConnStats.acksRetry += 1
                sendChunk(idx)
                sentNow += 1
            }
            val deadline = System.currentTimeMillis() + 3500
            while (System.currentTimeMillis() < deadline) {
                if (client?.isOpen != true) throw IllegalStateException("ws_closed")
                val msg = incoming.poll(80, TimeUnit.MILLISECONDS) ?: continue
                handleControl(msg)
                if (acks.size >= total && (0 until total).none { acks[it] != true }) return
            }
        }
    }

    private fun drainControl() {
        while (true) {
            val msg = incoming.poll() ?: return
            handleControl(msg)
        }
    }

    private fun handleControl(msg: JSONObject) {
        when (msg.optString("type")) {
            "resume" -> applyResume(msg)
            "chunk_nack" -> {
                val idx = msg.optInt("index")
                acks.remove(idx)
                sendChunk(idx)
            }
            "public_key" -> {
                if (!gotKey && msg.has("public_key")) {
                    crypto.derive(msg.getString("public_key"))
                    gotKey = true
                }
            }
            "transfer_response" -> {
                if (!msg.optBoolean("accepted")) throw RejectedException()
                accepted = true
            }
        }
    }

    private fun applyResume(msg: JSONObject) {
        val bitsB64 = msg.optString("have_bits")
        if (bitsB64.isNotBlank()) {
            val raw = Base64.decode(bitsB64, Base64.NO_WRAP)
            var i = 0
            while (i < total) {
                val on = i < raw.size * 8 && (raw[i shr 3].toInt() and (1 shl (i and 7))) != 0
                if (on) acks[i] = true else acks.remove(i)
                i++
            }
        }
        val have = msg.optJSONArray("have") ?: JSONArray()
        for (i in 0 until have.length()) acks[have.optInt(i)] = true
        val miss = msg.optJSONArray("missing")
        if (miss != null) {
            for (i in 0 until miss.length()) acks.remove(miss.optInt(i))
        }
        AppLog.i("resume have=${have.length()} missing=${miss?.length() ?: -1} bits=${bitsB64.isNotBlank()}")
        sendJson(
            JSONObject()
                .put("type", "resume_ack")
                .put("target_id", target)
                .put("next", (0 until total).firstOrNull { acks[it] != true } ?: total)
        )
    }

    private fun sendChunk(index: Int) {
        val ws = client ?: throw IllegalStateException("ws_closed")
        val chunk = source.read(index, CHUNK)
        if (chunk.isEmpty()) return
        val nonce = CryptoEngine.randomNonce()
        val enc = crypto.encrypt(chunk, nonce)
        ackAt[index] = System.currentTimeMillis()
        val end = index.toLong() * CHUNK + chunk.size
        ws.send(
            JSONObject()
                .put("type", "chunk")
                .put("target_id", target)
                .put("index", index)
                .put("data", Base64.encodeToString(enc, Base64.NO_WRAP))
                .put("nonce", Base64.encodeToString(nonce, Base64.NO_WRAP))
                .put("chunk_hash", sha256(chunk))
                .put("is_last", end >= fileSize)
                .toString()
        )
    }

    private fun sendJson(o: JSONObject) {
        val ws = client ?: throw IllegalStateException("ws_closed")
        ws.send(o.toString())
    }

    private fun dropSocket() {
        try {
            client?.close()
        } catch (_: Exception) {
        }
        client = null
    }

    private fun waitFor(type: String, sec: Long): JSONObject {
        val held = ArrayList<JSONObject>()
        val end = System.currentTimeMillis() + sec * 1000
        try {
            while (System.currentTimeMillis() < end) {
                val msg = incoming.poll(1, TimeUnit.SECONDS) ?: continue
                if (msg.optString("type") == type) return msg
                held.add(msg)
            }
            throw IllegalStateException("timeout")
        } finally {
            held.forEach { incoming.offer(it) }
        }
    }

    private class RejectedException : RuntimeException("rejected")

    companion object {
        const val CHUNK = 49152
        private fun sha256(data: ByteArray): String {
            val d = MessageDigest.getInstance("SHA-256").digest(data)
            return d.joinToString("") { "%02x".format(it) }
        }
    }
}

package com.p2psecure.share

import android.app.NotificationChannel
import android.app.NotificationManager
import android.app.PendingIntent
import android.app.Service
import android.content.Intent
import android.os.Build

class TransferService : Service() {
    @Volatile private var running = false
    private var wakeLock: PowerManager.WakeLock? = null

    override fun onBind(intent: Intent?): IBinder? = null

    override fun onCreate() {
        super.onCreate()
        if (Build.VERSION.SDK_INT >= 26) {
            val nm = getSystemService(NotificationManager::class.java)
            nm.createNotificationChannel(
                NotificationChannel("transfer", getString(R.string.channel_name), NotificationManager.IMPORTANCE_LOW)
            )
        }
        val open = PendingIntent.getActivity(
            this, 0, Intent(this, MainActivity::class.java),
            PendingIntent.FLAG_IMMUTABLE or PendingIntent.FLAG_UPDATE_CURRENT
        )
        val n = NotificationCompat.Builder(this, "transfer")
            .setContentTitle(getString(R.string.app_name))
            .setContentText(getString(R.string.notif_running))
            .setSmallIcon(R.mipmap.ic_launcher)
            .setContentIntent(open)
            .setOngoing(true)
            .build()
        startForeground(7, n)
        try {
            val pm = getSystemService(POWER_SERVICE) as PowerManager
            wakeLock = pm.newWakeLock(PowerManager.PARTIAL_WAKE_LOCK, "p2p:tunnel").also {
                it.setReferenceCounted(false)
                it.acquire()
            }
        } catch (e: Exception) {
            AppLog.w("wakelock ${e.message}")
        }
        running = true
        Thread {
            while (running) {
                try {
                    ShareRuntime.ensureStarted(applicationContext)
                    AppLog.i("service link=${ShareRuntime.receiverLink} err=${ShareRuntime.lastError}")
                    if (ShareRuntime.hasInternetLink()) {
                        Thread.sleep(8000)
                    } else {
                        Thread.sleep(4000)
                    }
                } catch (_: InterruptedException) {
                    break
                } catch (_: Exception) {
                    try {
                        Thread.sleep(6000)
                    } catch (_: Exception) {
                    }
                }
            }
        }.apply {
            name = "p2p-service"
            isDaemon = true
            start()
        }
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int = START_STICKY

    override fun onDestroy() {
        running = false
        try {
            wakeLock?.release()
        } catch (_: Exception) {
        }
        wakeLock = null
        super.onDestroy()
        ShareRuntime.stop()
    }
}

        val langSpinner = findViewById<Spinner>(R.id.langSpinner)
        val themeSpinner = findViewById<Spinner>(R.id.themeSpinner)
        val tokenBox = findViewById<EditText>(R.id.tokenBox)

        val themes = arrayOf("dark", "blue", "green")
        langSpinner.adapter = ArrayAdapter(this, android.R.layout.simple_spinner_dropdown_item, langs)
        themeSpinner.adapter = ArrayAdapter(this, android.R.layout.simple_spinner_dropdown_item, themes)
        langSpinner.setSelection(langs.indexOf(prefs.language).coerceAtLeast(0))
        themeSpinner.setSelection(themes.indexOf(prefs.theme).coerceAtLeast(0))
        tokenBox.setText(prefs.ngrokToken)
        findViewById<Button>(R.id.privacyBtn).setOnClickListener {
            val html = assets.open("privacy_policy.html").bufferedReader().readText()
            androidx.appcompat.app.AlertDialog.Builder(this)
                .setTitle(R.string.privacy)
                .setMessage(html.replace(Regex("<[^>]+>"), " ").trim())
                .setPositiveButton(android.R.string.ok, null)
                .show()
        }
        findViewById<Button>(R.id.saveBtn).setOnClickListener {
            prefs.language = langSpinner.selectedItem as String
            prefs.theme = themeSpinner.selectedItem as String
            prefs.ngrokToken = tokenBox.text.toString().trim()
            startActivity(
                Intent(this, MainActivity::class.java)
                    .addFlags(Intent.FLAG_ACTIVITY_NEW_TASK or Intent.FLAG_ACTIVITY_CLEAR_TASK)
            )
            finish()
        }
    }
}

package com.p2psecure.share

import android.content.Context
import org.json.JSONObject
import java.io.File
import java.net.HttpURLConnection
import java.net.URL
import java.util.concurrent.TimeUnit

class NgrokHelper(private val context: Context) {
    private var process: Process? = null
    var publicUrl: String = ""
        private set
    var lastError: String = ""
        private set

    fun install(): Boolean = true

    fun start(token: String, port: Int): String {
        return try {
            stop()
            lastError = ""
            publicUrl = ""
            if (token.isBlank()) {
                lastError = "missing_token"
                return ""
            }
            val bin = binaryFile() ?: return ""
            val home = context.filesDir
            fun proc(vararg args: String): Process {
                val pb = ProcessBuilder(listOf(bin.absolutePath) + args)
                    .directory(home)
                    .redirectErrorStream(true)
                pb.environment()["HOME"] = home.absolutePath
                pb.environment()["TMPDIR"] = context.cacheDir.absolutePath
                return pb.start()
            }
            try {
                proc("config", "add-authtoken", token).waitFor(8, TimeUnit.SECONDS)
            } catch (_: Exception) {
            }
            process = proc("http", port.toString(), "--authtoken", token, "--log=stdout")
            for (i in 0 until 12) {
                Thread.sleep(400)
                val url = apiUrl()
                if (url.isNotBlank()) {
                    publicUrl = url
                    return url
                }
                if (process?.isAlive == false) break
            }
            lastError = "tunnel_timeout"
            stop()
            ""
        } catch (e: Exception) {
            lastError = e.message ?: "start_failed"
            stop()
            ""
        }
    }

    fun stop() {
        try {
            process?.destroy()
        } catch (_: Exception) {
        }
        process = null
    }

    private fun binaryFile(): File? {
        val native = File(context.applicationInfo.nativeLibraryDir, "libngrok.so")
        if (native.exists() && native.length() > 1000) return native
        lastError = "native_missing"
        return null
    }

    private fun apiUrl(): String {
        return try {
            val conn = URL("http://127.0.0.1:4040/api/tunnels").openConnection() as HttpURLConnection
            conn.connectTimeout = 800
            conn.readTimeout = 800
            val body = conn.inputStream.bufferedReader().readText()
            val arr = JSONObject(body).getJSONArray("tunnels")
            for (i in 0 until arr.length()) {
                val u = arr.getJSONObject(i).optString("public_url")
                if (u.startsWith("https://")) return u
            }
            ""
        } catch (_: Exception) {
            ""
        }
    }
}

const BACKUP_WS = [];
const BLOCKED = ['.exe','.dll','.bat','.cmd','.sh','.ps1','.vbs','.js','.jar','.apk','.ipa','.scr'];
const CHUNK_DEFAULT = 49152;
const IDB_NAME = 'p2p-share';
const IDB_STORE = 'transfers';

function allWs() {
    const proto = location.protocol === 'https:' ? 'wss:' : 'ws:';
    const primary = proto + '//' + location.host + '/ws';
    const list = [];
    const q = new URLSearchParams(location.search).get('ws');
    if (q) list.push(q);
    list.push(primary);
    BACKUP_WS.forEach(u => { if (u && list.indexOf(u) < 0) list.push(u); });
    return list;
}

async function getDeviceId() {
    const data = [navigator.userAgent, navigator.language, screen.width, screen.height, navigator.hardwareConcurrency];
    const buf = await crypto.subtle.digest('SHA-256', new TextEncoder().encode(data.join('|')));
    return Array.from(new Uint8Array(buf)).map(b => b.toString(16).padStart(2,'0')).join('').slice(0,16);
}

async function genKeyPair() {
    return crypto.subtle.generateKey({name:'ECDH', namedCurve:'P-256'}, true, ['deriveKey', 'deriveBits']);
}

async function parseJwk(raw) {
    if (!raw) return null;
    return typeof raw === 'string' ? JSON.parse(raw) : raw;
}

async function deriveShared(priv, peerJwk) {
    const peer = await crypto.subtle.importKey('jwk', peerJwk, {name:'ECDH', namedCurve:'P-256'}, false, []);
    const bits = await crypto.subtle.deriveBits({name:'ECDH', public:peer}, priv, 256);
    const hkdfKey = await crypto.subtle.importKey('raw', bits, 'HKDF', false, ['deriveKey']);
    return crypto.subtle.deriveKey(
        {name:'HKDF', hash:'SHA-256', salt: new Uint8Array(32), info: new TextEncoder().encode('p2p-v1')},
        hkdfKey,
        {name:'AES-GCM', length:256},
        false,
        ['decrypt']
    );
}

function bitGet(i) {
    return !!(haveBits && i < totalChunks && (haveBits[i >> 3] & (1 << (i & 7))));
}
function bitSet(i) {
    if (!haveBits || i < 0 || i >= totalChunks) return false;
    const mask = 1 << (i & 7);
    if (haveBits[i >> 3] & mask) return false;
    haveBits[i >> 3] |= mask;
    gotCount += 1;
    return true;
}
function missingCount() { return totalChunks - gotCount; }
function sampleMissing(max) {
    const miss = [];
    for (let i = 0; i < totalChunks && miss.length < max; i++) if (!bitGet(i)) miss.push(i);
    return miss;
}
function bitsB64() {
    let bin = '';
    for (let i = 0; i < haveBits.length; i++) bin += String.fromCharCode(haveBits[i]);
    return btoa(bin);
}

function openDb() {
    return new Promise((resolve, reject) => {
        const req = indexedDB.open(IDB_NAME, 1);
        req.onupgradeneeded = () => req.result.createObjectStore(IDB_STORE);
        req.onsuccess = () => resolve(req.result);
        req.onerror = () => reject(req.error);
    });
}

            chunk_size: chunkSize,

            bits: haveBits,

}

}

async function openSink(name, size) {
    if (window.showSaveFilePicker) {
        try {
            const handle = await window.showSaveFilePicker({ suggestedName: name });
            const w = await handle.createWritable({ keepExistingData: true });
            sink = { kind: 'fs', writable: w, handle };
            sinkKind = 'fs';
            log('حفظ مباشر على القرص', 'ok');
            return;
        } catch (e) {
            if (e && e.name === 'AbortError') throw e;
            log('File System غير متاح — OPFS', 'info');
        }
    }
    if (navigator.storage && navigator.storage.getDirectory) {
        const root = await navigator.storage.getDirectory();
        const fh = await root.getFileHandle('p2p-' + (pending.file_hash || name), { create: true });
        let access;
        try { access = await fh.createSyncAccessHandle(); } catch (e) { access = null; }
        if (access) {
            sink = { kind: 'opfs-sync', handle: fh, access };
            sinkKind = 'opfs-sync';
        } else {
            sink = { kind: 'opfs', handle: fh };
            sinkKind = 'opfs';
        }
        log('حفظ مؤقت OPFS', 'ok');
        return;
    }
    if (size > 32 * 1024 * 1024) {
        throw new Error('no_disk_sink');
    }
    sink = { kind: 'mem', map: new Map() };
    sinkKind = 'mem';
    log('ذاكرة مؤقتة — ملف صغير فقط', 'info');
}

async function writeAt(index, bytes) {
    const offset = index * chunkSize;
    if (sink.kind === 'fs') {
        await sink.writable.write({ type: 'write', position: offset, data: bytes });
    } else if (sink.kind === 'opfs-sync') {
        sink.access.write(bytes, { at: offset });
    } else if (sink.kind === 'opfs') {
        const w = await sink.handle.createWritable({ keepExistingData: true });
        await w.write({ type: 'write', position: offset, data: bytes });
        await w.close();
    } else {
        sink.map.set(index, bytes);
    }
}

async function finishSink() {
    if (sink.kind === 'fs') {
        try { await sink.writable.close(); } catch (e) {}
        return 'disk';
    }
    if (sink.kind === 'opfs-sync') {
        try { sink.access.flush(); sink.access.close(); } catch (e) {}
        const file = await sink.handle.getFile();
        triggerDownload(file, pending.filename);
        try { const root = await navigator.storage.getDirectory(); await root.removeEntry(sink.handle.name); } catch (e) {}
        return 'opfs';
    }
    if (sink.kind === 'opfs') {
        const file = await sink.handle.getFile();
        triggerDownload(file, pending.filename);
        try { const root = await navigator.storage.getDirectory(); await root.removeEntry(sink.handle.name); } catch (e) {}
        return 'opfs';
    }
    const size = pending.file_size;
    const all = new Uint8Array(size);
    for (const [i, b] of sink.map.entries()) all.set(b, i * chunkSize);
    triggerDownload(new Blob([all]), pending.filename);
    sink.map.clear();
    return 'mem';
}

function triggerDownload(data, name) {
    const url = URL.createObjectURL(data instanceof Blob ? data : new Blob([data]));
    const a = document.createElement('a');
    a.href = url; a.download = name;
    document.body.appendChild(a); a.click();
    document.body.removeChild(a);
    setTimeout(() => URL.revokeObjectURL(url), 4000);
    log('تم التحميل: ' + name, 'ok');
}

async function connect() {
    if (connecting) return;
    connecting = true;
    deviceId = deviceId || await getDeviceId();
    document.getElementById('deviceId').textContent = deviceId;
    if (!kpReady) {
        kpReady = await genKeyPair();
        window.kp = kpReady;
    }
    const kp = kpReady;
    const list = allWs();
    const url = list[wsIndex % list.length];
    log('الاتصال: ' + url);
    try {
        ws = new WebSocket(url);
    } catch (err) {
        connecting = false;
        wsIndex += 1;
        setTimeout(connect, 1200);
        return;
    }

    ws.onopen = async () => {
        connecting = false;
        setStatus(receiving ? 'جاري الاستلام...' : 'متصل — في انتظار ملف...', receiving ? 'receiving' : 'connected');
        log('تم الاتصال بالخادم');
        const pub = await crypto.subtle.exportKey('jwk', kp.publicKey);
        ws.send(JSON.stringify({type:'register', mode:'receiver', device_id:deviceId, public_key:JSON.stringify(pub)}));
        if (pending && pending.from_id && acceptedOnce) sendResume();
    };

    ws.onmessage = async (e) => {
        try {
            const msg = JSON.parse(e.data);
            await handle(msg);
        } catch (err) {
            log('خطأ في معالجة الرسالة: ' + err, 'err');
        }
    };

    ws.onclose = () => {
        connecting = false;
        setStatus('انقطع الاتصال — تحويل المسار...', 'error');
        log('انقطع الاتصال — مسار آخر', 'err');
        wsIndex += 1;
        setTimeout(connect, 800);
    };

    ws.onerror = () => log('خطأ في الاتصال', 'err');
}

async function handle(msg) {
    switch(msg.type) {
        case 'registered':
            log('تم التسجيل: ' + msg.device_id, 'ok');
            break;
        case 'public_key':
            try {
                const pub = await parseJwk(msg.public_key);
                cryptoKey = await deriveShared(window.kp.privateKey, pub);
                log('تم تبادل المفاتيح', 'ok');
            } catch (err) {
                log('فشل تجهيز المفتاح: ' + err, 'err');
            }
            break;
        case 'transfer_request':
            await onRequest(msg);
            break;
        case 'chunk':
            await onChunk(msg);
            break;
        case 'transfer_complete':
            await onComplete(msg);
            break;
        case 'resume_ack':
            log('استئناف من القطعة ' + (msg.next || 0), 'ok');
            break;
        case 'pong': break;
        default: log('رسالة: ' + msg.type);
    }
}

async function onRequest(msg) {
    const ext = (msg.filename || '').slice(msg.filename.lastIndexOf('.')).toLowerCase();
    if (BLOCKED.includes(ext)) {
        ws.send(JSON.stringify({type:'transfer_response', target_id:msg.from_id, accepted:false}));
        log('نوع ملف ممنوع: ' + ext, 'err');
        return;
    }
    const same = pending && pending.filename === msg.filename && pending.file_hash === msg.file_hash;
    if ((receiving || acceptedOnce) && same) {
        pending.from_id = msg.from_id;
        const pub = await crypto.subtle.exportKey('jwk', window.kp.publicKey);
        ws.send(JSON.stringify({type:'public_key', target_id:msg.from_id, public_key: JSON.stringify(pub)}));
        ws.send(JSON.stringify({type:'transfer_response', target_id:msg.from_id, accepted:true, public_key: JSON.stringify(pub), resume:true}));
        sendResume();
        return;
    }
    pending = msg;
    totalChunks = msg.total_chunks || 0;
    chunkSize = msg.chunk_size || CHUNK_DEFAULT;
    haveBits = new Uint8Array(Math.ceil(totalChunks / 8));
    gotCount = 0;
    acceptedOnce = false;
    try {
        if (msg.public_key && !cryptoKey) {
            cryptoKey = await deriveShared(window.kp.privateKey, await parseJwk(msg.public_key));
            log('تم تبادل المفاتيح', 'ok');
        }
    } catch (err) {
        log('فشل تجهيز المفتاح: ' + err, 'err');
    }
    document.getElementById('fileName').textContent = msg.filename;
    document.getElementById('fileSize').textContent = fmt(msg.file_size);
    document.getElementById('senderId').textContent = msg.from_id;
    document.getElementById('request').classList.add('active');
    setStatus('طلب استلام ملف', 'waiting');
    log('طلب وارد: ' + msg.filename + ' (' + fmt(msg.file_size) + ')');
}

async function accept() {
    try {
        await openSink(pending.filename, pending.file_size);
    } catch (e) {
        log(e && e.message === 'no_disk_sink' ? 'المتصفح لا يدعم حفظ الملفات الكبيرة' : 'ألغيت عملية الحفظ', 'err');
        return;
    }
    const pub = await crypto.subtle.exportKey('jwk', window.kp.publicKey);
    const pubStr = JSON.stringify(pub);
    ws.send(JSON.stringify({type:'public_key', target_id:pending.from_id, public_key: pubStr}));
    ws.send(JSON.stringify({type:'transfer_response', target_id:pending.from_id, accepted:true, public_key: pubStr}));
    document.getElementById('request').classList.remove('active');
    document.getElementById('progress').classList.add('active');
    receiving = true;
    acceptedOnce = true;
    await saveMeta();
    setStatus('جاري الاستلام...', 'receiving');
    log('تم القبول — مرة واحدة للملف');
}

function reject() {
    ws.send(JSON.stringify({type:'transfer_response', target_id:pending.from_id, accepted:false}));
    document.getElementById('request').classList.remove('active');
    setStatus('تم الرفض', 'error');
    log('تم الرفض');
}

async function onChunk(msg) {
    if (!cryptoKey) { log('المفتاح غير جاهز', 'err'); return; }
    if (!sink) { log('مسار الحفظ غير جاهز', 'err'); return; }
    try {
        const enc = base64ToBuf(msg.data);
        const nonce = base64ToBuf(msg.nonce);
        const plain = await crypto.subtle.decrypt({name:'AES-GCM', iv:nonce}, cryptoKey, enc);
        const hash = await crypto.subtle.digest('SHA-256', plain);
        if (bufToHex(hash) !== msg.chunk_hash) {
            log('قطعة تالفة ' + msg.index + ' — طلب إعادة', 'err');
            ws.send(JSON.stringify({type:'chunk_nack', target_id:pending.from_id, index:msg.index}));
            return;
        }
        const bytes = new Uint8Array(plain);
        writeQueue = writeQueue.then(async () => {
            if (bitGet(msg.index)) return;
            await writeAt(msg.index, bytes);
            bitSet(msg.index);
            updateProgress();
            if (gotCount % 32 === 0) saveMeta();
        }).catch((e) => log('كتابة ' + e, 'err'));
        await writeQueue;
        ws.send(JSON.stringify({type:'chunk_ack', target_id:pending.from_id, index:msg.index}));
        if (msg.is_last) log('آخر قطعة وصلت');
    } catch(e) {
        log('خطأ في فك تشفير القطعة ' + msg.index + ': ' + e, 'err');
        if (pending && pending.from_id) {
            ws.send(JSON.stringify({type:'chunk_nack', target_id:pending.from_id, index:msg.index}));
        }
    }
}

function sendResume() {
    if (!ws || ws.readyState !== 1 || !pending || !pending.from_id) return;
    const miss = sampleMissing(64);
    ws.send(JSON.stringify({
        type:'resume',
        target_id: pending.from_id,
        filename: pending.filename,
        file_hash: pending.file_hash,
        have_bits: bitsB64(),
        missing: miss,
        total_chunks: totalChunks
    }));
    log('استئناف — ناقص ' + missingCount() + ' قطعة', 'info');
}

function updateProgress() {
    const pct = totalChunks ? (gotCount / totalChunks) * 100 : 0;
    const gotBytes = Math.min(gotCount * chunkSize, pending ? pending.file_size : 0);
    document.getElementById('progressFill').style.width = pct + '%';
    document.getElementById('progressText').textContent =
        pct.toFixed(1) + '% — ' + fmt(gotBytes) + ' / ' + fmt(pending ? pending.file_size : 0);
}

async function onComplete(msg) {
    await writeQueue;
    const miss = missingCount();
    if (miss) {
        log('نواقص ' + miss + ' — استئناف', 'err');
        sendResume();
        return;
    }
    setStatus('جاري إنهاء الحفظ...', 'receiving');
    await finishSink();
    await clearMeta();
    setStatus('اكتمل بنجاح', 'done');
    log('الملف سليم — تم الحفظ', 'ok');
    document.getElementById('progress').classList.remove('active');
    receiving = false;
    acceptedOnce = false;
    sink = null;
}

function fmt(b) {
    if (!b) return '0 B';
    const u = ['B','KB','MB','GB'];
    const i = Math.floor(Math.log(b)/Math.log(1024));
    return (b/Math.pow(1024,i)).toFixed(2) + ' ' + u[i];
}

function base64ToBuf(b64) {
    const bin = atob(b64);
    const u8 = new Uint8Array(bin.length);
    for (let i=0; i<bin.length; i++) u8[i] = bin.charCodeAt(i);
    return u8.buffer;
}

function bufToHex(buf) {
    return Array.from(new Uint8Array(buf)).map(b => b.toString(16).padStart(2,'0')).join('');
}

async function restoreSession() {
    const meta = await loadMeta();
    if (!meta || !meta.file_hash) return;
    pending = {
        filename: meta.filename,
        file_hash: meta.file_hash,
        file_size: meta.file_size,
        total_chunks: meta.total_chunks,
        chunk_size: meta.chunk_size
    };
    totalChunks = meta.total_chunks || 0;
    chunkSize = meta.chunk_size || CHUNK_DEFAULT;
    if (meta.bits instanceof Uint8Array) haveBits = meta.bits;
    else if (meta.bits instanceof ArrayBuffer) haveBits = new Uint8Array(meta.bits);
    else haveBits = new Uint8Array(meta.bits || Math.ceil(totalChunks / 8));
    gotCount = 0;
    for (let i = 0; i < totalChunks; i++) if (bitGet(i)) gotCount += 1;
    try {
        if (meta.fileHandle) {
            const fh = meta.fileHandle;
            let perm = 'granted';
            try { perm = await fh.queryPermission({ mode: 'readwrite' }); } catch (e) {}
            if (perm !== 'granted') {
                try { perm = await fh.requestPermission({ mode: 'readwrite' }); } catch (e) { perm = 'denied'; }
            }
            if (perm === 'granted') {
                const w = await fh.createWritable({ keepExistingData: true });
                sink = { kind: 'fs', writable: w, handle: fh };
                sinkKind = 'fs';
            }
        }
        if (!sink && navigator.storage && navigator.storage.getDirectory) {
            const root = await navigator.storage.getDirectory();
            const fh = await root.getFileHandle('p2p-' + meta.file_hash);
            let access = null;
            try { access = await fh.createSyncAccessHandle(); } catch (e) {}
            sink = access ? { kind: 'opfs-sync', handle: fh, access } : { kind: 'opfs', handle: fh };
            sinkKind = sink.kind;
        }
        if (!sink) return;
        acceptedOnce = true;
        receiving = true;
        document.getElementById('fileName').textContent = meta.filename;
        document.getElementById('fileSize').textContent = fmt(meta.file_size);
        document.getElementById('progress').classList.add('active');
        updateProgress();
        log('استعادة نقل سابق — ' + gotCount + '/' + totalChunks, 'ok');
    } catch (e) {
        log('تعذر استعادة الملف المؤقت', 'err');
    }
}
